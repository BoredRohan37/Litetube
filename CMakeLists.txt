cmake_minimum_required(VERSION 3.5)
project(LiteTube VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

find_package(Qt5 5.9 COMPONENTS Widgets Network REQUIRED)

# libmpv: pass -DMPV_ROOT=<dir containing include/mpv/client.h and the import library>
find_path(MPV_INCLUDE_DIR mpv/client.h HINTS ${MPV_ROOT} ${MPV_ROOT}/include)
find_library(MPV_LIBRARY NAMES mpv libmpv mpv-2 HINTS ${MPV_ROOT} ${MPV_ROOT}/lib ${MPV_ROOT}/lib64)
if(NOT MPV_INCLUDE_DIR OR NOT MPV_LIBRARY)
  message(FATAL_ERROR "libmpv not found. Set -DMPV_ROOT=... (see docs/BUILD.md)")
endif()

option(LITETUBE_SELFTEST "Build the headless self-test (--selftest)" OFF)
# CONFIGURE_DEPENDS (CMake 3.12+): new sources are picked up without a manual cmake re-run.
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.12)
  file(GLOB_RECURSE SRC CONFIGURE_DEPENDS src/*.cpp src/*.h)
else()
  file(GLOB_RECURSE SRC src/*.cpp src/*.h)
endif()
if(LITETUBE_SELFTEST)
  list(APPEND SRC tests/selftest.cpp)
endif()
if(WIN32)
  add_executable(LiteTube WIN32 ${SRC})
  target_link_libraries(LiteTube PRIVATE psapi)
  target_compile_definitions(LiteTube PRIVATE _WIN32_WINNT=0x0603 WINVER=0x0603 PSAPI_VERSION=1)
else()
  add_executable(LiteTube ${SRC})
endif()
target_include_directories(LiteTube PRIVATE src ${MPV_INCLUDE_DIR})
target_link_libraries(LiteTube PRIVATE Qt5::Widgets Qt5::Network ${MPV_LIBRARY})
target_compile_definitions(LiteTube PRIVATE LITETUBE_VERSION="${PROJECT_VERSION}")
if(LITETUBE_SELFTEST)
  target_compile_definitions(LiteTube PRIVATE LITETUBE_SELFTEST)
endif()
